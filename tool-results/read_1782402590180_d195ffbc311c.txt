   255→function getAccentForLabel(label: string): string {
   256→  if (/people/i.test(label)) return "bg-people";
   257→  if (/commits/i.test(label)) return "bg-muted-foreground/40";
   258→  if (/chunks/i.test(label)) return "bg-primary";
   259→  if (/skills/i.test(label)) return "bg-sector";
   260→  if (/diverse/i.test(label)) return "bg-amber-500";
   261→  if (/sector/i.test(label)) return "bg-sector";
   262→  return "bg-primary";
   263→}

   265→/** Compute and display which skills tend to co-occur in the same chunks.
   266→ *  Uses the per-person `allTags` flat list — any two tags that appear for the
   267→ *  same person+repo are considered co-occurring. */
   268→function SkillCoOccurrenceCard({ skillMap }: { skillMap: AdvancedSkillMap }) {
   269→  const pairs = useMemo(() => {
   270→    // Group all tags by (person, repo) to find co-occurrences
   271→    const byKey = new Map<string, { name: string; dimension: SkillDimension }[]>();
   272→    for (const p of skillMap.people) {
   273→      for (const t of p.allTags) {
   274→        const key = `${p.login}::${t.repo}`;
   275→        if (!byKey.has(key)) byKey.set(key, []);
   276→        byKey.get(key)!.push({ name: t.name, dimension: t.dimension });
   277→      }
   278→    }
   279→
   280→    // Count pairs (only across different dimensions to avoid trivial self-pairs)
   281→    const pairCounts = new Map<string, { a: string; b: string; count: number }>();
   282→    for (const tags of byKey.values()) {
   283→      const unique = Array.from(new Set(tags.map((t) => t.name)));
   284→      for (let i = 0; i < unique.length; i++) {
   285→        for (let j = i + 1; j < unique.length; j++) {
   286→          const [a, b] = [unique[i], unique[j]].sort();
   287→          const key = `${a}||${b}`;
   288→          if (!pairCounts.has(key)) {
   289→            pairCounts.set(key, { a, b, count: 0 });
   290→          }
   291→          pairCounts.get(key)!.count++;
   292→        }
   293→      }
   294→    }
   295→
   296→    return Array.from(pairCounts.values())
   297→      .filter((p) => p.count >= 2)
   298→      .sort((a, b) => b.count - a.count)
   299→      .slice(0, 20);
   300→  }, [skillMap]);
   301→
   302→  if (pairs.length === 0) return null;
   303→
   304→  const maxCount = Math.max(1, ...pairs.map((p) => p.count));
   305→
   306→  return (
   307→    <Card>
   308→      <CardHeader>
   309→        <CardTitle className="flex items-center gap-2 text-base">
   310→          <Link2 className="h-4 w-4 text-primary" />
   311→          Skill Co-occurrence
   312→        </CardTitle>
   313→        <CardDescription className="text-[11px] flex items-center gap-1">
   314→          <TrendingUp className="h-3 w-3" />
   315→          Skills that frequently appear together in the same person's work — reveals tech-stack clusters and cross-cutting capabilities
   316→        </CardDescription>
   317→      </CardHeader>
   318→      <CardContent className="space-y-1.5 max-h-[320px] overflow-y-auto pr-1">
   319→        {pairs.map((p, i) => (
   320→          <div key={`${p.a}-${p.b}`} className="flex items-center gap-3 text-[11px]">
   321→            <span className="font-mono text-muted-foreground w-5">{i + 1}</span>
   322→            <div className="flex-1 min-w-0 flex items-center gap-1.5">
   323→              <span className="font-medium truncate">{p.a}</span>
   324→              <Link2 className="h-2.5 w-2.5 text-muted-foreground shrink-0" />
   325→              <span className="font-medium truncate">{p.b}</span>
   326→            </div>
   327→            <span className="text-muted-foreground font-mono text-[10px] whitespace-nowrap">
   328→              {p.count}× co-occur
   329→            </span>
   330→            <div className="w-16 h-1.5 rounded-full bg-muted overflow-hidden shrink-0">
   331→              <div
   332→                className="h-full bg-primary"
   333→                style={{ width: `${(p.count / maxCount) * 100}%` }}
   334→              />
   335→            </div>
   336→          </div>
   337→        ))}
   338→      </CardContent>
   339→    </Card>
   340→  );
   341→}

   343→function LeaderboardCard({
   344→  title,
   345→  icon,
   346→  accent,
   347→  items,
   348→}: {
   349→  title: string;
   350→  icon: React.ReactNode;
   351→  accent: string;
   352→  items: { name: string; score: number; commits: number; people: number }[];
   353→}) {
   354→  const max = Math.max(1, ...items.map((i) => i.commits));
   355→  const totalCommits = items.reduce((sum, i) => sum + i.commits, 0);
   356→  return (
   357→    <Card>
   358→      <CardHeader className="pb-3">
   359→        <CardTitle className="flex items-center gap-2 text-base">
   360→          {icon}
   361→          {title}
   362→        </CardTitle>
   363→        <CardDescription className="text-[11px] flex items-center gap-1">
   364→          <TrendingUp className="h-3 w-3" /> {items.length} tags · {totalCommits} total commits
   365→        </CardDescription>
   366→      </CardHeader>
   367→      <CardContent className="space-y-1.5 max-h-[280px] overflow-y-auto pr-1">
   368→        {items.length === 0 && (
   369→          <div className="text-sm text-muted-foreground text-center py-4">No tags yet.</div>
   370→        )}
   371→        {items.map((item, i) => {
   372→          const pctOfMax = (item.commits / max) * 100;
   373→          const pctOfTotal = totalCommits > 0 ? (item.commits / totalCommits) * 100 : 0;
   374→          return (
   375→            <div
   376→              key={item.name}
   377→              className="group flex items-center gap-2 text-[11px] py-1 px-1 rounded hover:bg-muted/40 -mx-1 transition-colors"
   378→            >
   379→              <span className="font-mono text-muted-foreground w-4 text-right tabular-nums">{i + 1}</span>
   380→              <span className="font-medium flex-1 truncate" title={item.name}>
   381→                {item.name}
   382→              </span>
   383→              <span className="text-muted-foreground font-mono text-[10px] whitespace-nowrap tabular-nums">
   384→                {item.people}p · {item.commits}c
   385→              </span>
   386→              <span className="text-muted-foreground/80 font-mono text-[9px] w-9 text-right tabular-nums">
   387→                {pctOfTotal.toFixed(0)}%
   388→              </span>
   389→              <div className="w-20 h-2 rounded-full bg-muted overflow-hidden shrink-0 relative">
   390→                <div
   391→                  className={cn("h-full transition-all group-hover:brightness-110", accent)}
   392→                  style={{ width: `${pctOfMax}%` }}
   393→                />
   394→              </div>
   395→            </div>
   396→          );
   397→        })}
   398→      </CardContent>
   399→    </Card>
   400→  );

   403→/** Interactive heatmap of people × top skills across a chosen dimension.
   404→ *  Each cell shows commit volume for that (person, skill) pair, with color
   405→ *  intensity scaled to the max in the matrix. Clicking a row/column header
   406→ *  pins it for closer inspection. */
   407→function SkillCoverageMatrix({ skillMap }: { skillMap: AdvancedSkillMap }) {
   408→  const [dimension, setDimension] = useState<SkillDimension>("sector");
   409→  const [pinnedSkill, setPinnedSkill] = useState<string | null>(null);
   410→
   411→  const { people, skills, matrix, maxCell } = useMemo(() => {
   412→    // Pick the right org-wide list for the active dimension
   413→    const orgList =
   414→      dimension === "sector" ? skillMap.orgSectors :
   415→      dimension === "problemType" ? skillMap.orgProblemTypes :
   416→      dimension === "tech" ? skillMap.orgTech :
   417→      dimension === "methodology" ? skillMap.orgMethodologies : skillMap.orgRoles;
   418→
   419→    const skills = orgList.slice(0, 12).map((s) => s.name);
   420→
   421→    // Build matrix[personIndex][skillIndex] = commit count
   422→    const people = skillMap.people.slice(0, 12);
   423→    const matrix: number[][] = people.map((p) => {
   424→      const list =
   425→        dimension === "sector" ? p.sectors :
   426→        dimension === "problemType" ? p.problemTypes :
   427→        dimension === "tech" ? p.tech :
   428→        dimension === "methodology" ? p.methodologies : p.roles;
   429→      const lookup = new Map(list.map((s) => [s.name, s.commits]));
   430→      return skills.map((sk) => lookup.get(sk) ?? 0);
   431→    });
   432→
   433→    const maxCell = Math.max(1, ...matrix.flat());
   434→    return { people, skills, matrix, maxCell };
   435→  }, [skillMap, dimension]);
   436→
   437→  if (people.length === 0 || skills.length === 0) return null;
   438→
   439→  const accentVar =
   440→    dimension === "sector" ? "var(--sector)" :
   441→    dimension === "problemType" ? "var(--problem)" :
   442→    dimension === "tech" ? "var(--tech)" :
   443→    dimension === "methodology" ? "var(--methodology)" : "var(--role)";
   444→
   445→  const dimLabel =
   446→    dimension === "problemType" ? "Problem Types" : `${dimension.charAt(0).toUpperCase() + dimension.slice(1)}s`;
   447→
   448→  return (
   449→    <Card>
   450→      <CardHeader className="pb-3">
   451→        <div className="flex flex-wrap items-start justify-between gap-3">
   452→          <div>
   453→            <CardTitle className="flex items-center gap-2 text-base">
   454→              <Layers className="h-4 w-4 text-primary" />
   455→              Skill Coverage Matrix
   456→            </CardTitle>
   457→            <CardDescription className="text-[11px] flex items-center gap-1 mt-1">
   458→              <TrendingUp className="h-3 w-3" />
   459→              Heatmap of {people.length} top contributors × top {skills.length} {dimLabel.toLowerCase()} · darker = more commits
   460→            </CardDescription>
   461→          </div>
   462→          <Tabs value={dimension} onValueChange={(v) => { setDimension(v as SkillDimension); setPinnedSkill(null); }}>
   463→            <TabsList className="h-8">
   464→              {([
   465→                { k: "sector", l: "Sector" },
   466→                { k: "problemType", l: "Problem" },
   467→                { k: "tech", l: "Tech" },
   468→                { k: "methodology", l: "Method" },
   469→                { k: "role", l: "Role" },
   470→              ] as { k: SkillDimension; l: string }[]).map((d) => (
   471→                <TabsTrigger key={d.k} value={d.k} className="text-[10px] px-2 h-7">
   472→                  {d.l}
   473→                </TabsTrigger>
   474→              ))}
   475→            </TabsList>
   476→          </Tabs>
   477→        </div>
   478→      </CardHeader>
   479→      <CardContent className="overflow-x-auto">
   480→        <table className="w-full border-separate border-spacing-1">
   481→          <thead>
   482→            <tr>
   483→              <th className="sticky left-0 z-10 bg-card p-1.5 text-left text-[10px] font-medium text-muted-foreground min-w-[120px]">
   484→                Person
   485→              </th>
   486→              {skills.map((s) => (
   487→                <th
   488→                  key={s}
   489→                  className={cn(
   490→                    "p-1.5 text-[9px] font-medium cursor-pointer transition-colors rounded select-none",
   491→                    pinnedSkill === s ? "bg-primary/15 text-primary" : "text-muted-foreground hover:text-foreground"
   492→                  )}
   493→                  onClick={() => setPinnedSkill(pinnedSkill === s ? null : s)}
   494→                  title={`Click to pin ${s}`}
   495→                >
   496→                  <div className="writing-vertical-rl rotate-180 truncate max-h-[80px] mx-auto" style={{ writingMode: "vertical-rl" }}>
   497→                    {s}
   498→                  </div>
   499→                </th>
   500→              ))}
   501→            </tr>
   502→          </thead>
   503→          <tbody>
   504→            {people.map((p, ri) => (
   505→              <tr key={p.login}>

   579→/** Person-to-Person similarity matrix — NxN grid of overall Jaccard similarity.
   580→ *  Helps spot skill-cluster overlaps at a glance. Hover a cell to see exact %,
   581→ *  click an off-diagonal cell to open the Skill Graph in compare mode with
   582→ *  that pair pre-selected (requires onComparePair). */
   583→function PersonSimilarityMatrix({
   584→  skillMap,
   585→  onComparePair,
   586→}: {
   587→  skillMap: AdvancedSkillMap;
   588→  onComparePair?: (loginA: string, loginB: string) => void;
   589→}) {
   590→  const people = skillMap.people.slice(0, 10);
   591→  const [hovered, setHovered] = useState<{ i: number; j: number } | null>(null);
   592→
   593→  // Precompute Jaccard for every pair
   594→  const matrix = useMemo(() => {
   595→    const dimLists = (p: PersonSkillRecord) => [
   596→      ...p.sectors, ...p.problemTypes, ...p.tech, ...p.methodologies, ...p.roles,
   597→    ].map((s) => s.name);
   598→    const nameSets = people.map((p) => new Set(dimLists(p)));
   599→    return people.map((_, i) =>
   600→      people.map((_, j) => {
   601→        if (i === j) return { jaccard: 1, shared: nameSets[i].size, union: nameSets[i].size };
   602→        const a = nameSets[i];
   603→        const b = nameSets[j];
   604→        let shared = 0;
   605→        for (const n of a) if (b.has(n)) shared++;
   606→        const union = a.size + b.size - shared;
   607→        return { jaccard: union > 0 ? shared / union : 0, shared, union };
   608→      })
   609→    );
   610→  }, [people]);
   611→
   612→  if (people.length < 2) return null;
   613→

   615→    if (j >= 0.7) return "oklch(0.70 0.13 145)";
   616→    if (j >= 0.4) return "oklch(0.75 0.10 145)";
   617→    if (j >= 0.2) return "oklch(0.82 0.06 95)";
   618→    if (j > 0) return "oklch(0.80 0.07 50)";
   619→    return "var(--muted)";
   620→  };

   624→      <CardHeader className="pb-3">

   627→            <CardTitle className="flex items-center gap-2 text-base">

   631→            <CardDescription className="text-[11px] flex items-center gap-1 mt-1">
   632→              <TrendingUp className="h-3 w-3" />
   633→              Jaccard similarity across all 5 dimensions · darker green = more overlap · {people.length}×{people.length} grid

   635→          </div>
   636→        </div>
   637→      </CardHeader>
   638→      <CardContent className="overflow-x-auto">
   639→        <table className="border-separate border-spacing-1 mx-auto">
   640→          <thead>
   641→            <tr>
   642→              <th className="p-1.5"></th>
   643→              {people.map((p, j) => (
   644→                <th key={p.login} className="p-1 text-[9px] font-medium text-muted-foreground align-bottom">
   645→                  <div className="writing-vertical-rl rotate-180 mx-auto max-h-[70px] truncate" style={{ writingMode: "vertical-rl" }} title={p.name || p.login}>
   646→                    {p.name || p.login}
   647→                  </div>
   648→                </th>
   649→              ))}
   650→            </tr>
   651→          </thead>
   652→          <tbody>
   653→            {people.map((p, i) => (
   654→              <tr key={p.login}>
   655→                <td className="p-1 text-[10px] font-medium text-right whitespace-nowrap pr-2 max-w-[110px]">
   656→                  <div className="flex items-center gap-1.5 justify-end">
   657→                    <span className="truncate max-w-[90px]" title={p.name || p.login}>{p.name || p.login}</span>
   658→                    <Avatar className="h-4 w-4 shrink-0">
   659→                      <AvatarImage src={p.avatarUrl} />
   660→                      <AvatarFallback className="text-[7px]">{p.login[0]?.toUpperCase()}</AvatarFallback>
   661→                    </Avatar>
   662→                  </div>
   663→                </td>
   664→                {people.map((_, j) => {
   665→                  const v = matrix[i][j];
   666→                  const isDiag = i === j;
   667→                  const isHovered = hovered?.i === i && hovered?.j === j;
   668→                  const canClick = !isDiag && onComparePair;
   669→                  return (
   670→                    <td
   671→                      key={j}
   672→                      className="p-0 text-center align-middle"
   673→                      onMouseEnter={() => setHovered({ i, j })}
   674→                      onMouseLeave={() => setHovered(null)}
   675→                      title={isDiag
   676→                        ? `${p.name || p.login} (self)`
   677→                        : `${p.name || p.login} vs ${people[j].name || people[j].login}: ${(v.jaccard * 100).toFixed(0)}% (${v.shared}/${v.union} shared)${canClick ? " · click to compare" : ""}`}
   678→                    >
   679→                      <button
   680→                        type="button"
   681→                        disabled={!canClick}
   682→                        onClick={() => {
   683→                          if (canClick) onComparePair(p.login, people[j].login);
   684→                        }}
   685→                        className={cn(
   686→                          "h-9 w-9 rounded flex items-center justify-center text-[10px] font-mono font-semibold tabular-nums transition-all",

  1040→      dimKeys.forEach(({ key }) => {
  1041→        byDim[key] = Math.round((rawByDim[key][personIdx] / maxByDim[key]) * 100);
  1042→      });
  1043→      return byDim;
  1044→    });
  1045→    return { scores };
  1046→  }, [people]);

  1088→              <TrendingUp className="h-3 w-3" />

  1102→              className="max-w-full"
