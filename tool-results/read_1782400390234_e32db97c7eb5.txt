   459→  );

   489→function PeopleTable({
   490→  skillMap,
   491→  onSwitchToGraph,
   492→}: {
   493→  skillMap: AdvancedSkillMap;
   494→  onSwitchToGraph: (login: string) => void;
   495→}) {
   496→  const maxCommits = Math.max(1, ...skillMap.people.map((p) => p.totalCommits));
   497→
   498→  const exportCsv = () => {
   499→    const escape = (s: string) => `"${(s ?? "").replace(/"/g, '""')}"`;
   500→    const joinSkills = (list: { name: string }[]) => list.map((s) => s.name).join(" | ");
   501→    const header = [
   502→      "name", "login", "commits", "chunks", "repos",
   503→      "sectors", "problem_types", "tech", "methodologies", "roles",
   504→      "avatar_url",
   505→    ].join(",");
   506→    const rows = skillMap.people.map((p) => [
   507→      escape(p.name || p.login),
   508→      escape(p.login),
   509→      p.totalCommits,
   510→      p.totalChunks,
   511→      p.repos.length,
   512→      escape(joinSkills(p.sectors)),
   513→      escape(joinSkills(p.problemTypes)),
   514→      escape(joinSkills(p.tech)),
   515→      escape(joinSkills(p.methodologies)),
   516→      escape(joinSkills(p.roles)),
   517→      escape(p.avatarUrl),
   518→    ].join(","));
   519→    const csv = [header, ...rows].join("\n");
   520→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   521→    const url = URL.createObjectURL(blob);
   522→    const a = document.createElement("a");
   523→    a.href = url;

   637→type SkillVariant = "sector" | "problem" | "tech" | "methodology" | "role";
   638→
   639→function SkillChipList({
   640→  items,
   641→  variant,
   642→  max = 3,
   643→}: {
   644→  items: { name: string }[];
   645→  variant: SkillVariant;
   646→  max?: number;
   647→}) {
   648→  if (items.length === 0) {
   649→    return <div className="text-[10px] text-muted-foreground/40 italic">—</div>;
   650→  }
   651→  const shown = items.slice(0, max);
   652→  const overflow = items.length - shown.length;
   653→  // Stronger contrast: opaque text + tinted bg + slightly darker border
   654→  const cls: Record<SkillVariant, string> = {
   655→    sector: "border-sector/40 text-sector bg-sector/10",
   656→    problem: "border-problem/40 text-problem bg-problem/10",
   657→    tech: "border-tech/40 text-tech bg-tech/10",
   658→    methodology: "border-methodology/40 text-methodology bg-methodology/10",
   659→    role: "border-role/40 text-role bg-role/10",
   660→  };
   661→  return (
   662→    <div className="flex flex-wrap gap-1 max-w-[260px]">
   663→      {shown.map((s) => (
   664→        <span
   665→          key={s.name}
   666→          className={cn(
   667→            "inline-block text-[10px] px-1.5 py-0.5 rounded border font-medium leading-tight",
   668→            cls[variant]
   669→          )}
   670→          title={s.name}
   671→        >
   672→          {s.name}
   673→        </span>
   674→      ))}
   675→      {overflow > 0 && (
   676→        <span
   677→          className="inline-block text-[10px] px-1.5 py-0.5 rounded border border-foreground/20 bg-muted text-foreground/80 font-mono font-semibold tabular-nums leading-tight"
   678→          title={`${overflow} more: ${items.slice(max).map((s) => s.name).join(", ")}`}
   679→        >
   680→          +{overflow}
   681→        </span>
   682→      )}
   683→    </div>
   684→  );
   685→}
