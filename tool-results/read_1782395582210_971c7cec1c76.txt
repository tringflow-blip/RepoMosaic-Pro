     2→
     3→import { useMemo, useState } from "react";
     4→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
     5→import { Badge } from "@/components/ui/badge";
     6→import { Button } from "@/components/ui/button";
     7→import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar";
     8→import { Input } from "@/components/ui/input";
     9→import { Tabs, TabsList, TabsTrigger } from "@/components/ui/tabs";
    10→import {
    11→  Users,
    12→  Search,
    13→  GitCommitVertical,
    14→  FolderGit2,
    15→  Boxes,
    16→  Wrench,
    17→  Target,
    18→  Compass,
    19→  Shield,
    20→  ChevronRight,

    34→type Props = {
    35→  skillMap: AdvancedSkillMap;

    38→const DIMENSIONS: { key: SkillDimension; label: string; icon: typeof Compass; color: string }[] = [
    39→  { key: "sector", label: "Sectors", icon: Compass, color: "text-sector" },
    40→  { key: "problemType", label: "Problem Types", icon: Target, color: "text-problem" },
    41→  { key: "tech", label: "Tech", icon: Wrench, color: "text-tech" },
    42→  { key: "methodology", label: "Methodology", icon: Boxes, color: "text-methodology" },
    43→  { key: "role", label: "Roles", icon: Shield, color: "text-role" },
    44→];
    45→
    46→export function AdvancedSkillGraph({ skillMap }: Props) {
    47→  const [dimension, setDimension] = useState<SkillDimension>("sector");
    48→  const [search, setSearch] = useState("");
    49→  const [selectedLogin, setSelectedLogin] = useState<string | null>(null);
    50→  const [compareMode, setCompareMode] = useState(false);
    51→  const [compareLogins, setCompareLogins] = useState<string[]>([]);
    52→
    53→  const graph = useMemo(() => skillMapToGraph(skillMap, dimension), [skillMap, dimension]);
    54→
    55→  const filteredPeople = useMemo(() => {
    56→    const q = search.toLowerCase().trim();
    57→    if (!q) return skillMap.people;
    58→    return skillMap.people.filter(
    59→      (p) =>
    60→        p.login.toLowerCase().includes(q) ||
    61→        (p.name ?? "").toLowerCase().includes(q) ||
    62→        p.sectors.some((s) => s.name.toLowerCase().includes(q)) ||
    63→        p.problemTypes.some((s) => s.name.toLowerCase().includes(q)) ||
    64→        p.tech.some((s) => s.name.toLowerCase().includes(q)) ||
    65→        p.methodologies.some((s) => s.name.toLowerCase().includes(q)) ||
    66→        p.roles.some((s) => s.name.toLowerCase().includes(q))
    67→    );
    68→  }, [skillMap.people, search]);
    69→
    70→  const selectedPerson = selectedLogin
    71→    ? skillMap.people.find((p) => p.login === selectedLogin) ?? null
    72→    : null;
    73→
    74→  const comparePeople = compareLogins
    75→    .map((l) => skillMap.people.find((p) => p.login === l))
    76→    .filter((p): p is PersonSkillRecord => !!p);
    77→
    78→  const handlePersonClick = (login: string) => {
    79→    if (compareMode) {
    80→      setCompareLogins((prev) =>
    81→        prev.includes(login)
    82→          ? prev.filter((l) => l !== login)
    83→          : prev.length >= 2
    84→            ? [prev[1], login] // replace first, keep second
    85→            : [...prev, login]
    86→      );
    87→    } else {
    88→      setSelectedLogin(selectedLogin === login ? null : login);
    89→    }
    90→  };
    91→
    92→  return (
    93→    <div className="space-y-4">
    94→      {/* Compare mode banner */}
    95→      {compareMode && (
    96→        <Card className="border-primary/30 bg-primary/5">
    97→          <CardContent className="py-3 flex flex-wrap items-center justify-between gap-3">
    98→            <div className="flex items-center gap-2 text-xs">
    99→              <GitCompare className="h-4 w-4 text-primary" />
   100→              <span className="font-medium">Compare mode</span>
   101→              <span className="text-muted-foreground">
   102→                · Select {comparePeople.length === 0 ? "2" : comparePeople.length === 1 ? "1 more" : "✓ 2 selected"} people from the list
   103→              </span>
   104→            </div>
   105→            <div className="flex items-center gap-2">
   106→              {comparePeople.map((p) => (
   107→                <Badge key={p.login} variant="outline" className="text-[10px] gap-1">
   108→                  <Avatar className="h-3 w-3">
   109→                    <AvatarImage src={p.avatarUrl} />
   110→                    <AvatarFallback className="text-[8px]">{p.login[0]?.toUpperCase()}</AvatarFallback>
   111→                  </Avatar>
   112→                  {p.name || p.login}
   113→                  <button
   114→                    type="button"
   115→                    onClick={() => setCompareLogins((prev) => prev.filter((l) => l !== p.login))}
   116→                    className="ml-0.5 hover:text-destructive"
   117→                    aria-label={`Remove ${p.login}`}
   118→                  >
   119→                    <X className="h-2.5 w-2.5" />
   120→                  </button>
   121→                </Badge>
   122→              ))}
   123→              <Button
   124→                size="sm"
   125→                variant="ghost"
   126→                className="h-7 text-[11px]"
   127→                onClick={() => { setCompareMode(false); setCompareLogins([]); }}
   128→              >
   129→                Exit compare
   130→              </Button>
   131→            </div>
   132→          </CardContent>
   133→        </Card>
   134→      )}
   135→
   136→      <div className="grid gap-6 lg:grid-cols-5">
   137→        <div className="lg:col-span-3 space-y-4">
   138→          <Card>
   139→            <CardHeader>
   140→              <div className="flex flex-wrap items-center justify-between gap-3">
   141→                <div>
   142→                  <CardTitle className="flex items-center gap-2">
   143→                    <Users className="h-4 w-4 text-people" />
   144→                    Skill Graph — {skillMap.org}
   145→                  </CardTitle>
   146→                  <CardDescription className="mt-1">
   147→                    {skillMap.people.length} people · {skillMap.totalRepos} repos ·{" "}
   148→                    {skillMap.totalCommits} commits · {skillMap.totalChunks} LLM chunks ·{" "}
   149→                    <span className="font-mono text-[10px]">{skillMap.model}</span>
   150→                  </CardDescription>
   151→                </div>
   152→                <div className="flex items-center gap-2">
   153→                  <Button
   154→                    size="sm"
   155→                    variant={compareMode ? "default" : "outline"}
   156→                    className="h-7 text-[11px]"
   157→                    onClick={() => {
   158→                      setCompareMode(!compareMode);
   159→                      setCompareLogins([]);
   160→                      setSelectedLogin(null);
   161→                    }}
   162→                  >
   163→                    <GitCompare className="h-3 w-3 mr-1" />
   164→                    Compare
   165→                  </Button>
   166→                  <Tabs value={dimension} onValueChange={(v) => setDimension(v as SkillDimension)}>
   167→                    <TabsList className="h-8">
   168→                      {DIMENSIONS.map((d) => (
   169→                        <TabsTrigger key={d.key} value={d.key} className="text-[11px] px-2.5 h-7">
   170→                          <d.icon className={cn("h-3 w-3 mr-1", d.color)} />
   171→                          {d.label}
   172→                        </TabsTrigger>
   173→                      ))}
   174→                    </TabsList>
   175→                  </Tabs>
   176→                </div>
   177→              </div>
   178→            </CardHeader>
   179→          <CardContent>
   180→            <ForceGraph
   181→              nodes={graph.nodes}
   182→              edges={graph.edges}
   183→              height={520}
   184→              showLabels
   185→              emptyMessage="Run a scan to populate the skill graph."
   186→              onSelectNode={(id) => {
   187→                if (id.startsWith("person:")) {
   188→                  const login = id.slice("person:".length);
   189→                  handlePersonClick(login);
   190→                }
   191→              }}
   192→            />
   193→            <div className="flex items-center justify-between gap-3 mt-2">
   194→              <p className="text-[11px] text-muted-foreground">
   195→                {compareMode
   196→                  ? "Compare mode: click person nodes to select 2 for side-by-side comparison."
   197→                  : "Person nodes (people-colored) sized by commits · "}
   198→                {!compareMode && (
   199→                  <>
   200→                    {DIMENSIONS.find((d) => d.key === dimension)?.label} nodes sized by aggregate commit
   201→                    volume · drag to reposition
   202→                  </>
   203→                )}
   204→              </p>
   205→              <DimensionLegend dimension={dimension} />
   206→            </div>
   207→          </CardContent>
   208→        </Card>
   209→
   210→        <DimensionLeaderboards skillMap={skillMap} dimension={dimension} />
   211→      </div>
   212→
   213→      <div className="lg:col-span-2 space-y-4">
   214→        <Card>
   215→          <CardHeader>
   216→            <CardTitle className="text-base flex items-center justify-between">
   217→              <span>Contributors</span>
   218→              {compareMode && (
   219→                <span className="text-[10px] font-normal text-muted-foreground">
   220→                  {compareLogins.length}/2 selected
   221→                </span>
   222→              )}
   223→            </CardTitle>
   224→            <div className="relative mt-2">
   225→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   226→              <Input
   227→                placeholder="Search by name, skill, sector…"
   228→                value={search}
   229→                onChange={(e) => setSearch(e.target.value)}
   230→                className="pl-8 text-sm h-9"
   231→              />
   232→            </div>
   233→          </CardHeader>
   234→          <CardContent className="space-y-1 max-h-[300px] overflow-y-auto pr-1">
   235→            {filteredPeople.slice(0, 80).map((p) => (
   236→              <PersonRow
   237→                key={p.login}
   238→                person={p}
   239→                selected={compareMode ? compareLogins.includes(p.login) : selectedLogin === p.login}
   240→                onClick={() => handlePersonClick(p.login)}
   241→                compareMode={compareMode}
   242→              />
   243→            ))}
   244→            {filteredPeople.length === 0 && (
   245→              <div className="text-sm text-muted-foreground py-6 text-center">No matches.</div>
   246→            )}
   247→          </CardContent>
   248→        </Card>
   249→
   250→        {compareMode ? (
   251→          comparePeople.length === 2 ? (
   252→            <ComparePeopleCard people={comparePeople as [PersonSkillRecord, PersonSkillRecord]} />
   253→          ) : (
   254→            <Card className="border-dashed border-primary/30">
   255→              <CardContent className="pt-6 text-center text-sm text-muted-foreground space-y-2">
   256→                <GitCompare className="h-6 w-6 mx-auto text-primary/50" />
   257→                <div>Select {2 - comparePeople.length} more person{comparePeople.length === 1 ? "" : "s"} to compare</div>
   258→                <div className="text-[11px]">Click contributor rows or graph nodes</div>
   259→              </CardContent>
   260→            </Card>
   261→          )
   262→        ) : selectedPerson ? (
   263→          <PersonDetailCard person={selectedPerson} />
   264→        ) : (
   265→          <Card className="border-dashed">
   266→            <CardContent className="pt-6 text-center text-sm text-muted-foreground">
   267→              Click a contributor to see their multi-dimensional skill breakdown.
   268→            </CardContent>
   269→          </Card>
   270→        )}
   271→        </div>
   272→      </div>
   273→    </div>
   274→  );
   275→}

   277→/** Side-by-side comparison of two people's skill profiles. Shows overlap
   278→ *  (shared skills) and unique skills per person, per dimension. */
   279→function ComparePeopleCard({ people }: { people: [PersonSkillRecord, PersonSkillRecord] }) {
   280→  const [a, b] = people;
   281→  const dims: { key: "sectors" | "problemTypes" | "tech" | "methodologies" | "roles"; label: string; color: string }[] = [
   282→    { key: "sectors", label: "Sectors", color: "text-sector" },
   283→    { key: "problemTypes", label: "Problem Types", color: "text-problem" },
   284→    { key: "tech", label: "Tech", color: "text-tech" },
   285→    { key: "methodologies", label: "Methodologies", color: "text-methodology" },
   286→    { key: "roles", label: "Roles", color: "text-role" },
   287→  ];
   288→
   289→  return (
   290→    <Card className="animate-fade-in-up">
   291→      <CardHeader>
   292→        <CardTitle className="flex items-center gap-2 text-base">
   293→          <GitCompare className="h-4 w-4 text-primary" />
   294→          Skill Overlap
   295→        </CardTitle>
   296→        <CardDescription className="text-[11px]">
   297→          Shared skills (both people) vs unique skills per person
   298→        </CardDescription>
   299→      </CardHeader>
   300→      <CardContent className="space-y-4 max-h-[560px] overflow-y-auto pr-1">
   301→        {/* Header row with both avatars */}
   302→        <div className="grid grid-cols-[1fr_auto_1fr] gap-2 items-center pb-2 border-b">
   303→          <div className="flex items-center gap-2 min-w-0">
   304→            <Avatar className="h-7 w-7 shrink-0">
   305→              <AvatarImage src={a.avatarUrl} />
   306→              <AvatarFallback className="text-[10px]">{a.login[0]?.toUpperCase()}</AvatarFallback>
   307→            </Avatar>
   308→            <div className="min-w-0">
   309→              <div className="text-xs font-medium truncate">{a.name || a.login}</div>
   310→              <div className="text-[10px] text-muted-foreground">{a.totalCommits}c · {a.totalChunks} chunks</div>
   311→            </div>
   312→          </div>
   313→          <VsBadge />
   314→          <div className="flex items-center gap-2 min-w-0 justify-end">
   315→            <div className="min-w-0 text-right">
   316→              <div className="text-xs font-medium truncate">{b.name || b.login}</div>
   317→              <div className="text-[10px] text-muted-foreground">{b.totalCommits}c · {b.totalChunks} chunks</div>
   318→            </div>
   319→            <Avatar className="h-7 w-7 shrink-0">
   320→              <AvatarImage src={b.avatarUrl} />
   321→              <AvatarFallback className="text-[10px]">{b.login[0]?.toUpperCase()}</AvatarFallback>
   322→            </Avatar>
   323→          </div>
   324→        </div>
   325→
   326→        {dims.map((dim) => {
   327→          const listA = a[dim.key];
   328→          const listB = b[dim.key];
   329→          const namesA = new Set(listA.map((s) => s.name));
   330→          const namesB = new Set(listB.map((s) => s.name));
   331→          const shared = listA.filter((s) => namesB.has(s.name));
   332→          const onlyA = listA.filter((s) => !namesB.has(s.name));
   333→          const onlyB = listB.filter((s) => !namesA.has(s.name));
   334→          if (shared.length === 0 && onlyA.length === 0 && onlyB.length === 0) return null;
   335→          return (
   336→            <div key={dim.key} className="space-y-1.5">
   337→              <div className={cn("text-[11px] font-medium flex items-center justify-between", dim.color)}>
   338→                <span>{dim.label}</span>
   339→                <span className="text-[9px] text-muted-foreground font-normal">
   340→                  {shared.length} shared · {onlyA.length}/{onlyB.length} unique
   341→                </span>
   342→              </div>
   343→              {shared.length > 0 && (
   344→                <div className="space-y-0.5">
   345→                  <div className="text-[9px] uppercase tracking-wide text-emerald-600 flex items-center gap-1">
   346→                    <CheckCircle2 className="h-2.5 w-2.5" /> Shared
   347→                  </div>
   348→                  {shared.map((s) => (
   349→                    <div key={s.name} className="flex items-center gap-2 text-[10px]">
   350→                      <span className="font-medium flex-1 truncate">{s.name}</span>
   351→                      <span className="text-muted-foreground font-mono">{s.commits}c</span>
   352→                      <span className="text-muted-foreground/60 font-mono">
   353→                        {b[dim.key].find((x) => x.name === s.name)?.commits ?? 0}c
   354→                      </span>
   355→                    </div>
   356→                  ))}
   357→                </div>
   358→              )}
   359→              <div className="grid grid-cols-2 gap-2">
   360→                <div className="space-y-0.5">
   361→                  <div className="text-[9px] uppercase tracking-wide text-muted-foreground truncate">
   362→                    Only {a.name || a.login}
   363→                  </div>
   364→                  {onlyA.slice(0, 5).map((s) => (
   365→                    <div key={s.name} className="text-[10px] truncate" title={s.name}>
   366→                      <span className="font-medium">{s.name}</span>
   367→                      <span className="text-muted-foreground ml-1 font-mono">{s.commits}c</span>
   368→                    </div>
   369→                  ))}
   370→                  {onlyA.length === 0 && <div className="text-[9px] text-muted-foreground/60">—</div>}
   371→                </div>
   372→                <div className="space-y-0.5">
   373→                  <div className="text-[9px] uppercase tracking-wide text-muted-foreground truncate text-right">
   374→                    Only {b.name || b.login}
   375→                  </div>
   376→                  {onlyB.slice(0, 5).map((s) => (
   377→                    <div key={s.name} className="text-[10px] truncate text-right" title={s.name}>
   378→                      <span className="text-muted-foreground mr-1 font-mono">{s.commits}c</span>
   379→                      <span className="font-medium">{s.name}</span>
   380→                    </div>
   381→                  ))}
   382→                  {onlyB.length === 0 && <div className="text-[9px] text-muted-foreground/60 text-right">—</div>}
   383→                </div>
   384→              </div>
   385→            </div>
   386→          );
   387→        })}
   388→      </CardContent>
   389→    </Card>
   390→  );
   391→}

   401→function PersonRow({
   402→  person,
   403→  selected,
   404→  onClick,
   405→  compareMode = false,
   406→}: {
   407→  person: PersonSkillRecord;
   408→  selected: boolean;
   409→  onClick: () => void;
   410→  compareMode?: boolean;
   411→}) {
   412→  return (
   413→    <button
   414→      onClick={onClick}
   415→      className={cn(
   416→        "w-full flex items-center gap-3 p-2 rounded-lg border transition-all text-left",
   417→        selected
   418→          ? compareMode
   419→            ? "border-primary/40 bg-primary/10 shadow-soft"
   420→            : "border-people/40 bg-accent/50 shadow-soft"
   421→          : "border-transparent hover:border-border/60 hover:bg-muted/40"
   422→      )}
   423→    >
   424→      {compareMode && (
   425→        <div className={cn(
   426→          "h-4 w-4 rounded-full border flex items-center justify-center shrink-0 transition-colors",
   427→          selected ? "border-primary bg-primary text-primary-foreground" : "border-muted-foreground/30"

   433→        <AvatarImage src={person.avatarUrl} />

   439→          <span>@{person.login}</span>
   440→          <span>·</span>
   441→          <span className="flex items-center gap-0.5">
   442→            <GitCommitVertical className="h-3 w-3" />
   443→            {person.totalCommits}
   444→          </span>
   445→          <span>·</span>
   446→          <span className="flex items-center gap-0.5">
   447→            <FolderGit2 className="h-3 w-3" />
   448→            {person.repos.length}
   449→          </span>
   450→        </div>
   451→      </div>
   452→      {person.sectors[0] && (
   453→        <Badge variant="outline" className="text-[10px] font-mono border-sector/30 text-sector">
   454→          {person.sectors[0].name}
   455→        </Badge>
   456→      )}
   457→    </button>
   458→  );
   459→}

   545→/** Skill list where each row can be expanded to show the commit-level evidence
   546→ *  that justified the tag. Evidence comes from the `allTags` flat list. */
   547→function SkillListWithEvidence({
   548→  title,
   549→  icon,
   550→  color,
   551→  barClass,
   552→  items,
   553→  allTags,
   554→  dimension,
   555→}: {
   556→  title: string;
   557→  icon: React.ReactNode;
   558→  color: string;
   559→  barClass: string;
   560→  items: { name: string; score: number; commits: number; chunks: number }[];
   561→  allTags: { dimension: SkillDimension; name: string; evidence: string[]; repo: string; commits: number }[];
   562→  dimension: SkillDimension;
   563→}) {
   564→  const [expanded, setExpanded] = useState<string | null>(null);
   565→  if (items.length === 0) return null;
   566→  return (
   567→    <div>
   568→      <div className={cn("text-xs font-medium mb-2 flex items-center gap-1.5", color)}>
   569→        {icon}
   570→        {title}
   571→      </div>
   572→      <div className="space-y-1">
   573→        {items.slice(0, 10).map((s) => {
   574→          const isOpen = expanded === s.name;
   575→          // Find evidence entries for this skill in this dimension
   576→          const evidenceEntries = allTags
   577→            .filter((t) => t.dimension === dimension && t.name === s.name)
   578→            .flatMap((t) => (t.evidence || []).map((e) => ({ evidence: e, repo: t.repo })))
   579→            .slice(0, 6);
   580→          return (
   581→            <div key={s.name} className="space-y-1">
   582→              <button
   583→                type="button"
   584→                onClick={() => setExpanded(isOpen ? null : s.name)}
   585→                className="w-full flex items-center justify-between text-[11px] group"
   586→              >
   587→                <span className="font-medium truncate pr-2 text-left flex items-center gap-1">
   588→                  {evidenceEntries.length > 0 && (
   589→                    <ChevronRight
   590→                      className={cn(
   591→                        "h-2.5 w-2.5 text-muted-foreground transition-transform shrink-0",
   592→                        isOpen && "rotate-90"
   593→                      )}
   594→                    />
   595→                  )}
   596→                  {s.name}
   597→                </span>
   598→                <span className="text-muted-foreground font-mono whitespace-nowrap flex items-center gap-1.5">
   599→                  <span>{(s.score * 100).toFixed(0)}% · {s.commits}c</span>
   600→                  {evidenceEntries.length > 0 && (
   601→                    <span className="text-[8px] uppercase tracking-wide text-muted-foreground/60 group-hover:text-muted-foreground">
   602→                      {evidenceEntries.length} evidence
   603→                    </span>
   604→                  )}
   605→                </span>
   606→              </button>
   607→              <div className="h-1 rounded-full bg-muted overflow-hidden">
   608→                <div className={cn("h-full transition-all", barClass)} style={{ width: `${s.score * 100}%` }} />
   609→              </div>
   610→              {isOpen && evidenceEntries.length > 0 && (
   611→                <div className="mt-1 mb-1.5 p-2 rounded-md bg-muted/40 border border-border/40 space-y-1 animate-fade-in-up">
   612→                  <div className="text-[9px] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1">
   613→                    <Search className="h-2.5 w-2.5" />
   614→                    Commit-level evidence from GLM
   615→                  </div>
   616→                  {evidenceEntries.map((e, i) => (
   617→                    <div key={i} className="text-[10px] flex items-start gap-1.5">
   618→                      <FolderGit2 className="h-2.5 w-2.5 mt-0.5 shrink-0 text-muted-foreground/60" />
   619→                      <span className="font-mono text-muted-foreground shrink-0">{e.repo}:</span>
   620→                      <span className="text-foreground/80 break-all">{e.evidence}</span>
   621→                    </div>
   622→                  ))}
   623→                </div>
   624→              )}
   625→            </div>
   626→          );
   627→        })}
   628→      </div>
   629→    </div>
   630→  );
   631→}

   633→function DimensionLeaderboards({
   634→  skillMap,
   635→  dimension,
   636→}: {
   637→  skillMap: AdvancedSkillMap;
   638→  dimension: SkillDimension;
   639→}) {
   640→  const list =
   641→    dimension === "sector" ? skillMap.orgSectors :
   642→    dimension === "problemType" ? skillMap.orgProblemTypes :
   643→    dimension === "tech" ? skillMap.orgTech :
   644→    dimension === "methodology" ? skillMap.orgMethodologies : skillMap.orgRoles;
   645→
   646→  if (list.length === 0) return null;
   647→
   648→  const maxCommits = Math.max(1, ...list.map((l) => l.commits));
   649→  return (
   650→    <Card>
   651→      <CardHeader>
   652→        <CardTitle className="text-base capitalize">
   653→          {dimension === "problemType" ? "Problem Types" : dimension + "s"} — Org-wide
   654→        </CardTitle>
   655→        <CardDescription className="text-[11px]">
   656→          Top {list.length} tags detected across all scanned repos
   657→        </CardDescription>
   658→      </CardHeader>
   659→      <CardContent className="space-y-1.5 max-h-[300px] overflow-y-auto pr-1">
   660→        {list.slice(0, 25).map((item, i) => (
   661→          <div key={item.name} className="flex items-center gap-3 text-[11px]">
   662→            <span className="font-mono text-muted-foreground w-6">{i + 1}</span>
   663→            <span className="font-medium flex-1 truncate">{item.name}</span>
   664→            <span className="text-muted-foreground font-mono whitespace-nowrap">
   665→              {item.people}p · {item.commits}c
   666→            </span>
   667→            <div className="w-24 h-1 rounded-full bg-muted overflow-hidden">
   668→              <div
   669→                className={cn(
   670→                  "h-full",
   671→                  dimension === "sector" && "bg-sector",
   672→                  dimension === "problemType" && "bg-problem",
   673→                  dimension === "tech" && "bg-tech",
   674→                  dimension === "methodology" && "bg-methodology",
   675→                  dimension === "role" && "bg-role"
   676→                )}
   677→                style={{ width: `${(item.commits / maxCommits) * 100}%` }}
   678→              />
   679→            </div>
   680→          </div>
   681→        ))}
   682→      </CardContent>
   683→    </Card>
   684→  );
   685→}
