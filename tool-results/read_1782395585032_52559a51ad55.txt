   283→   283→    { key: "problemTypes", label: "Problem Types", color: "text-problem" },

   288→   288→
   289→   289→  return (

   545→   545→/** Skill list where each row can be expanded to show the commit-level evidence
   546→   546→ *  that justified the tag. Evidence comes from the `allTags` flat list. */
   547→   547→function SkillListWithEvidence({
   548→   548→  title,
   549→   549→  icon,
   550→   550→  color,
   551→   551→  barClass,
   552→   552→  items,
   553→   553→  allTags,
   554→   554→  dimension,
   555→   555→}: {
   556→   556→  title: string;
   557→   557→  icon: React.ReactNode;
   558→   558→  color: string;
   559→   559→  barClass: string;
   560→   560→  items: { name: string; score: number; commits: number; chunks: number }[];
   561→   561→  allTags: { dimension: SkillDimension; name: string; evidence: string[]; repo: string; commits: number }[];
   562→   562→  dimension: SkillDimension;
   563→   563→}) {
   564→   564→  const [expanded, setExpanded] = useState<string | null>(null);
   565→   565→  if (items.length === 0) return null;
   566→   566→  return (
   567→   567→    <div>
   568→   568→      <div className={cn("text-xs font-medium mb-2 flex items-center gap-1.5", color)}>
   569→   569→        {icon}
   570→   570→        {title}
   571→   571→      </div>
   572→   572→      <div className="space-y-1">
   573→   573→        {items.slice(0, 10).map((s) => {
   574→   574→          const isOpen = expanded === s.name;
   575→   575→          // Find evidence entries for this skill in this dimension
   576→   576→          const evidenceEntries = allTags
   577→   577→            .filter((t) => t.dimension === dimension && t.name === s.name)
   578→   578→            .flatMap((t) => (t.evidence || []).map((e) => ({ evidence: e, repo: t.repo })))
   579→   579→            .slice(0, 6);
   580→   580→          return (
   581→   581→            <div key={s.name} className="space-y-1">
   582→   582→              <button
   583→   583→                type="button"
   584→   584→                onClick={() => setExpanded(isOpen ? null : s.name)}
   585→   585→                className="w-full flex items-center justify-between text-[11px] group"
   586→   586→              >
   587→   587→                <span className="font-medium truncate pr-2 text-left flex items-center gap-1">
   588→   588→                  {evidenceEntries.length > 0 && (
   589→   589→                    <ChevronRight
   590→   590→                      className={cn(
   591→   591→                        "h-2.5 w-2.5 text-muted-foreground transition-transform shrink-0",
   592→   592→                        isOpen && "rotate-90"
   593→   593→                      )}
   594→   594→                    />
   595→   595→                  )}
   596→   596→                  {s.name}
   597→   597→                </span>
   598→   598→                <span className="text-muted-foreground font-mono whitespace-nowrap flex items-center gap-1.5">
   599→   599→                  <span>{(s.score * 100).toFixed(0)}% · {s.commits}c</span>
   600→   600→                  {evidenceEntries.length > 0 && (
   601→   601→                    <span className="text-[8px] uppercase tracking-wide text-muted-foreground/60 group-hover:text-muted-foreground">
   602→   602→                      {evidenceEntries.length} evidence
   603→   603→                    </span>
   604→   604→                  )}
   605→   605→                </span>
   606→   606→              </button>
   607→   607→              <div className="h-1 rounded-full bg-muted overflow-hidden">
   608→   608→                <div className={cn("h-full transition-all", barClass)} style={{ width: `${s.score * 100}%` }} />
   609→   609→              </div>
   610→   610→              {isOpen && evidenceEntries.length > 0 && (
   611→   611→                <div className="mt-1 mb-1.5 p-2 rounded-md bg-muted/40 border border-border/40 space-y-1 animate-fade-in-up">
   612→   612→                  <div className="text-[9px] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1">
   613→   613→                    <Search className="h-2.5 w-2.5" />
   614→   614→                    Commit-level evidence from GLM
   615→   615→                  </div>
   616→   616→                  {evidenceEntries.map((e, i) => (
   617→   617→                    <div key={i} className="text-[10px] flex items-start gap-1.5">
   618→   618→                      <FolderGit2 className="h-2.5 w-2.5 mt-0.5 shrink-0 text-muted-foreground/60" />
   619→   619→                      <span className="font-mono text-muted-foreground shrink-0">{e.repo}:</span>
   620→   620→                      <span className="text-foreground/80 break-all">{e.evidence}</span>
   621→   621→                    </div>
   622→   622→                  ))}
   623→   623→                </div>
   624→   624→              )}
   625→   625→            </div>
   626→   626→          );
   627→   627→        })}
   628→   628→      </div>
   629→   629→    </div>
   630→   630→  );
