   394→/** Side-by-side comparison of two people's skill profiles. Shows overlap
   395→ *  (shared skills) and unique skills per person, per dimension. Includes
   396→ *  a Jaccard similarity score per dimension (|shared| / |union|) and an
   397→ *  overall similarity headline. */
   398→function ComparePeopleCard({ people }: { people: [PersonSkillRecord, PersonSkillRecord] }) {
   399→  const [a, b] = people;
   400→  const dims: { key: "sectors" | "problemTypes" | "tech" | "methodologies" | "roles"; label: string; color: string; barClass: string }[] = [
   401→    { key: "sectors", label: "Sectors", color: "text-sector", barClass: "bg-sector" },
   402→    { key: "problemTypes", label: "Problem Types", color: "text-problem", barClass: "bg-problem" },
   403→    { key: "tech", label: "Tech", color: "text-tech", barClass: "bg-tech" },
   404→    { key: "methodologies", label: "Methodologies", color: "text-methodology", barClass: "bg-methodology" },
   405→    { key: "roles", label: "Roles", color: "text-role", barClass: "bg-role" },
   406→  ];
   407→
   408→  // Compute per-dimension Jaccard + overall
   409→  const dimStats = dims.map((dim) => {
   410→    const listA = a[dim.key];
   411→    const listB = b[dim.key];
   412→    const namesA = new Set(listA.map((s) => s.name));
   413→    const namesB = new Set(listB.map((s) => s.name));
   414→    const shared = listA.filter((s) => namesB.has(s.name));
   415→    const onlyA = listA.filter((s) => !namesB.has(s.name));
   416→    const onlyB = listB.filter((s) => !namesA.has(s.name));
   417→    const union = namesA.size + namesB.size - shared.length;
   418→    const jaccard = union > 0 ? shared.length / union : 0;
   419→    return { dim, shared, onlyA, onlyB, jaccard, union };
   420→  });
   421→
   422→  // Overall Jaccard across all dimensions
   423→  const allA = new Set<string>();
   424→  const allB = new Set<string>();
   425→  for (const d of dims) {
   426→    a[d.key].forEach((s) => allA.add(s.name));
   427→    b[d.key].forEach((s) => allB.add(s.name));
   428→  }
   429→  let sharedAll = 0;
   430→  for (const name of allA) if (allB.has(name)) sharedAll++;
   431→  const overallJaccard = allA.size + allB.size - sharedAll > 0
   432→    ? sharedAll / (allA.size + allB.size - sharedAll)
   433→    : 0;
   434→
   435→  const jaccardLabel = (j: number) =>
   436→    j >= 0.7 ? "Very similar" : j >= 0.4 ? "Moderately similar" : j >= 0.2 ? "Somewhat similar" : j > 0 ? "Mostly distinct" : "No overlap";
   437→
   438→  return (
   439→    <Card className="animate-fade-in-up">
   440→      <CardHeader>
   441→        <CardTitle className="flex items-center gap-2 text-base">
   442→          <GitCompare className="h-4 w-4 text-primary" />
   443→          Skill Overlap
   444→        </CardTitle>
   445→        <CardDescription className="text-[11px]">
   446→          Shared skills vs unique skills per person, with Jaccard similarity (|shared| / |union|)
   447→        </CardDescription>
   448→      </CardHeader>
   449→      <CardContent className="space-y-4 max-h-[560px] overflow-y-auto pr-1">
   450→        {/* Overall similarity headline */}
   451→        <div className="rounded-lg border bg-gradient-to-br from-primary/5 to-transparent p-3 space-y-2">
   452→          <div className="flex items-center justify-between">
   453→            <span className="text-[10px] uppercase tracking-wide text-muted-foreground flex items-center gap-1">
   454→              <Sparkles className="h-3 w-3" />
   455→              Overall skill similarity
   456→            </span>
   457→            <span className={cn(
   458→              "text-[10px] font-mono font-semibold px-1.5 py-0.5 rounded",
   459→              overallJaccard >= 0.4 ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400" : "bg-amber-500/15 text-amber-700 dark:text-amber-400"
   460→            )}>
   461→              {jaccardLabel(overallJaccard)}
   462→            </span>
   463→          </div>
   464→          <div className="flex items-baseline gap-2">
   465→            <span className="text-3xl font-mono font-bold tabular-nums">
   466→              {(overallJaccard * 100).toFixed(0)}<span className="text-base text-muted-foreground">%</span>
   467→            </span>
   468→            <span className="text-[10px] text-muted-foreground">
   469→              {sharedAll} shared · {allA.size + allB.size - sharedAll} total unique
   470→            </span>
   471→          </div>
   472→          <div className="h-2 rounded-full bg-muted overflow-hidden">
   473→            <div
   474→              className="h-full bg-gradient-to-r from-primary/70 to-primary transition-all"
   475→              style={{ width: `${overallJaccard * 100}%` }}
   476→            />
   477→          </div>
   478→        </div>
   479→
   480→        {/* Header row with both avatars */}
   481→        <div className="grid grid-cols-[1fr_auto_1fr] gap-2 items-center pb-2 border-b">
   482→          <div className="flex items-center gap-2 min-w-0">
   483→            <Avatar className="h-7 w-7 shrink-0">
   484→              <AvatarImage src={a.avatarUrl} />
   485→              <AvatarFallback className="text-[10px]">{a.login[0]?.toUpperCase()}</AvatarFallback>
   486→            </Avatar>

   733→/** Skill list where each row can be expanded to show the commit-level evidence
   734→ *  that justified the tag. Evidence comes from the `allTags` flat list. */
   735→function SkillListWithEvidence({
   736→  title,
   737→  icon,
   738→  color,
   739→  barClass,
   740→  items,
   741→  allTags,
   742→  dimension,
   743→}: {
   744→  title: string;
   745→  icon: React.ReactNode;
   746→  color: string;
   747→  barClass: string;
   748→  items: { name: string; score: number; commits: number; chunks: number }[];
   749→  allTags: { dimension: SkillDimension; name: string; evidence: string[]; repo: string; commits: number }[];
   750→  dimension: SkillDimension;
   751→}) {
   752→  const [expanded, setExpanded] = useState<string | null>(null);
   753→  if (items.length === 0) return null;
   754→  return (
   755→    <div>
   756→      <div className={cn("text-xs font-medium mb-2 flex items-center gap-1.5", color)}>
   757→        {icon}
   758→        {title}
   759→      </div>
   760→      <div className="space-y-1">
   761→        {items.slice(0, 10).map((s) => {
   762→          const isOpen = expanded === s.name;
   763→          // Find evidence entries for this skill in this dimension
   764→          const evidenceEntries = allTags
   765→            .filter((t) => t.dimension === dimension && t.name === s.name)
   766→            .flatMap((t) => (t.evidence || []).map((e) => ({ evidence: e, repo: t.repo })))
   767→            .slice(0, 6);
   768→          return (
   769→            <div key={s.name} className="space-y-1">
   770→              <button
   771→                type="button"
   772→                onClick={() => setExpanded(isOpen ? null : s.name)}
   773→                className="w-full flex items-center justify-between text-[11px] group"
   774→              >
   775→                <span className="font-medium truncate pr-2 text-left flex items-center gap-1">
   776→                  {evidenceEntries.length > 0 && (
   777→                    <ChevronRight
   778→                      className={cn(
   779→                        "h-2.5 w-2.5 text-muted-foreground transition-transform shrink-0",
   780→                        isOpen && "rotate-90"
   781→                      )}
   782→                    />
   783→                  )}
   784→                  {s.name}
   785→                </span>
   786→                <span className="text-muted-foreground font-mono whitespace-nowrap flex items-center gap-1.5">
   787→                  <span>{(s.score * 100).toFixed(0)}% · {s.commits}c</span>
   788→                  {evidenceEntries.length > 0 && (
   789→                    <span className="text-[8px] uppercase tracking-wide text-muted-foreground/60 group-hover:text-muted-foreground">
   790→                      {evidenceEntries.length} evidence
   791→                    </span>
   792→                  )}
   793→                </span>
   794→              </button>
   795→              <div className="h-1 rounded-full bg-muted overflow-hidden">
   796→                <div className={cn("h-full transition-all", barClass)} style={{ width: `${s.score * 100}%` }} />
   797→              </div>
   798→              {isOpen && evidenceEntries.length > 0 && (
   799→                <div className="mt-1 mb-1.5 p-2 rounded-md bg-muted/40 border border-border/40 space-y-1 animate-fade-in-up">
   800→                  <div className="text-[9px] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1">
   801→                    <Search className="h-2.5 w-2.5" />
   802→                    Commit-level evidence from GLM
   803→                  </div>
   804→                  {evidenceEntries.map((e, i) => (
   805→                    <div key={i} className="text-[10px] flex items-start gap-1.5">
   806→                      <FolderGit2 className="h-2.5 w-2.5 mt-0.5 shrink-0 text-muted-foreground/60" />
   807→                      <span className="font-mono text-muted-foreground shrink-0">{e.repo}:</span>
   808→                      <span className="text-foreground/80 break-all">{e.evidence}</span>
   809→                    </div>
   810→                  ))}
   811→                </div>
   812→              )}
   813→            </div>
   814→          );
   815→        })}
   816→      </div>
   817→    </div>
   818→  );
   819→}
