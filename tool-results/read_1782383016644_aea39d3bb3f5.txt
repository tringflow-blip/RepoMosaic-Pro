   153→
   154→  // Theme

   160→
   161→  // Auth
   162→  const [token, setToken] = useState(settings.githubToken ?? "");
   163→  const [viewer, setViewer] = useState<Viewer | null>(null);
   164→  const [authLoading, setAuthLoading] = useState(false);
   165→
   166→  // Org

   168→  const [org, setOrg] = useState<Org | null>(null);
   169→  const [orgLoading, setOrgLoading] = useState(false);
   170→

   173→  const [repoFilter, setRepoFilter] = useState("");
   174→  const [selectedRepos, setSelectedRepos] = useState<Set<string>>(new Set());
   175→  const [showArchived, setShowArchived] = useState(false);
   176→  const [showForks, setShowForks] = useState(false);
   177→  const [reposLoading, setReposLoading] = useState(false);
   178→  const [branchesByRepo, setBranchesByRepo] = useState<Record<string, string[]>>({});
   179→  const [branchMode, setBranchMode] = useState<"main" | "all">(settings.defaultBranchMode);
   180→
   181→  // Scan
   182→  const [scanId, setScanId] = useState<string | null>(null);
   183→  const [scanProgress, setScanProgress] = useState<{
   184→    phase: string;
   185→    message: string;
   186→    progress: number;
   187→    totalRepos?: number;
   188→    processedRepos?: number;

   190→  const [scanResult, setScanResult] = useState<ScanResult | null>(null);
   191→  const [polling, setPolling] = useState(false);

   195→
   196→  // ---- Actions ----

   198→    if (!token.trim()) {
   199→      toast({ title: "Token required", variant: "destructive" });
   200→      return;
   201→    }
   202→    setAuthLoading(true);

   204→      const res = await fetch("/api/github/viewer", {
   205→        method: "POST",
   206→        headers: { "Content-Type": "application/json" },
   207→        body: JSON.stringify({ token }),
   208→      });
   209→      if (!res.ok) {
   210→        const e = await res.json();
   211→        throw new Error(e.error || "Auth failed");
   212→      }
   213→      const v = await res.json();
   214→      setViewer(v);
   215→      saveSettings({ githubToken: token });
   216→      toast({ title: `Signed in as ${v.login}` });
   217→    } catch (e) {
   218→      toast({
   219→        title: "Authentication failed",
   220→        description: e instanceof Error ? e.message : "Unknown error",
   221→        variant: "destructive",
   222→      });
   223→    } finally {
   224→      setAuthLoading(false);
   225→    }
   226→  }, [token, toast]);
   227→
   228→  const handleLoadOrg = useCallback(async () => {
   229→    if (!orgInput.trim() || !token) {
   230→      toast({ title: "Enter org URL and token first", variant: "destructive" });
   231→      return;
   232→    }
   233→    setOrgLoading(true);
   234→    setSelectedRepos(new Set());
   235→    setRepos([]);
   236→    setOrg(null);
   237→    setScanResult(null);
   238→    setScanId(null);
   239→    setScanProgress(null);
   240→    try {
   241→      const res = await fetch("/api/github/org", {
   242→        method: "POST",
   243→        headers: { "Content-Type": "application/json" },
   244→        body: JSON.stringify({ token, identifier: orgInput }),
   245→      });
   246→      if (!res.ok) {
   247→        const e = await res.json();
   248→        throw new Error(e.error || "Failed to load org");
   249→      }

   252→      // Auto-fetch repos
   253→      setReposLoading(true);

   255→        method: "POST",

   257→        body: JSON.stringify({ token, identifier: orgInput }),
   258→      });
   259→      if (!r2.ok) {
   260→        const e = await r2.json();
   261→        throw new Error(e.error || "Failed to load repos");
   262→      }
   263→      const r = await r2.json();
   264→      const filtered: Repo[] = (r.repos as Repo[]).filter(
   265→        (x) => (showArchived || !x.isArchived) && (showForks || !x.isFork)

   269→      setSelectedRepos(new Set(filtered.map((x) => x.name)));
   270→      // Fetch branches for each repo in parallel (limited)
   271→      const branches: Record<string, string[]> = {};
   272→      await Promise.all(
   273→        filtered.slice(0, 50).map(async (repo) => {
   274→          try {
   275→            const r3 = await fetch("/api/github/branches", {
   276→              method: "POST",
   277→              headers: { "Content-Type": "application/json" },
   278→              body: JSON.stringify({ token, owner: o.login, repo: repo.name }),
   279→            });
   280→            if (r3.ok) {
   281→              const b = await r3.json();
   282→              branches[repo.name] = b.branches.map((x: { name: string }) => x.name);
   283→            }
   284→          } catch {}
   285→        })
   286→      );
   287→      setBranchesByRepo(branches);
   288→      toast({ title: `Loaded ${filtered.length} repos from ${o.login}` });
   289→    } catch (e) {
   290→      toast({
   291→        title: "Failed to load org",
   292→        description: e instanceof Error ? e.message : "Unknown error",
   293→        variant: "destructive",
   294→      });
   295→    } finally {
   296→      setOrgLoading(false);
   297→      setReposLoading(false);
   298→    }
   299→  }, [orgInput, token, showArchived, showForks, toast]);
   300→

   302→    setSelectedRepos((cur) => {
   303→      const next = new Set(cur);
   304→      if (next.has(name)) next.delete(name);
   305→      else next.add(name);
   306→      return next;
   307→    });
   308→  };
   309→
   310→  const selectAll = () => setSelectedRepos(new Set(filteredRepos.map((r) => r.name)));
   311→  const deselectAll = () => setSelectedRepos(new Set());
   312→
   313→  const filteredRepos = repos.filter((r) => {
   314→    const q = repoFilter.toLowerCase();

   316→      !q ||
   317→      r.name.toLowerCase().includes(q) ||
   318→      (r.description ?? "").toLowerCase().includes(q) ||
   319→      (r.language ?? "").toLowerCase().includes(q)
   320→    );
   321→  });
   322→
   323→  // ---- Scan ----
   324→  const startScan = useCallback(async () => {

   326→      toast({ title: "Token, org, and folder are required", variant: "destructive" });
   327→      return;
   328→    }
   329→    if (selectedRepos.size === 0) {
   330→      toast({ title: "Select at least one repo", variant: "destructive" });
   331→      return;
   332→    }
   333→    setScanResult(null);
   334→    setScanProgress({ phase: "init", message: "Starting…", progress: 0 });
   335→    setActiveTab("scan");
   336→    try {
   337→      const res = await fetch("/api/scan/start", {
   338→        method: "POST",
   339→        headers: { "Content-Type": "application/json" },
   340→        body: JSON.stringify({
   341→          token,
   342→          identifier: orgInput,
   343→          selectedRepos: Array.from(selectedRepos),
   344→          branchMode,
   345→          workspacePath: workspace.folderPath,
   346→        }),
   347→      });
   348→      if (!res.ok) {
   349→        const e = await res.json();
   350→        throw new Error(e.error || "Failed to start scan");
   351→      }
   352→      const { scanId: sid } = await res.json();
   353→      setScanId(sid);
   354→      setPolling(true);
   355→    } catch (e) {
   356→      toast({
   357→        title: "Scan failed to start",
   358→        description: e instanceof Error ? e.message : "Unknown error",
   359→        variant: "destructive",
   360→      });
   361→      setScanProgress(null);
   362→    }
   363→  }, [token, org, workspace, selectedRepos, branchMode, orgInput, toast]);
   364→
   365→  // Poll scan status
   366→  useEffect(() => {
   367→    if (!scanId || !polling) return;

   370→      try {
   371→        const res = await fetch(`/api/scan/start?scanId=${scanId}`);
   372→        if (!res.ok) return;
   373→        const data = await res.json();
   374→        if (stopped) return;
   375→        setScanProgress({
   376→          phase: data.phase,
   377→          message: data.message,
   378→          progress: data.progress,
   379→          totalRepos: data.totalRepos,
   380→          processedRepos: data.processedRepos,
   381→        });

   383→          setPolling(false);
   384→          setScanResult(data.result as ScanResult);
   385→          setActiveTab("capability");
   386→          toast({ title: "Scan complete", description: `Analyzed ${data.totalRepos} repos` });
   387→        } else if (data.phase === "error") {
   388→          setPolling(false);
   389→          toast({
   390→            title: "Scan failed",
   391→            description: data.error,
   392→            variant: "destructive",
   393→          });
   394→        }
   395→      } catch {}
   396→    };
   397→    const interval = setInterval(poll, 1500);
   398→    poll();
   399→    return () => {
   400→      stopped = true;
   401→      clearInterval(interval);
   402→    };
   403→  }, [scanId, polling, toast]);
   404→

   406→
   407→  return (

   409→      {/* Header */}
   410→      <header className="sticky top-0 z-40 border-b border-border/60 bg-background/80 backdrop-blur-xl">
   411→        <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   412→          <div className="flex h-16 items-center justify-between gap-4">
   413→            <div className="flex items-center gap-3">
   414→              <div className="relative h-9 w-9 rounded-lg bg-primary text-primary-foreground grid place-items-center shadow-soft">
   415→                <Network className="h-5 w-5" strokeWidth={2.5} />
   416→              </div>
   417→              <div>
   418→                <div className="text-base font-semibold tracking-tight">RepoMosaic</div>

   420→                  Org Capability &amp; Skill Intelligence
   421→                </div>
   422→              </div>
   423→            </div>
   424→
   425→            <div className="flex items-center gap-2">
   426→              <Button
   427→                variant="ghost"
   428→                size="icon"
   429→                onClick={() => setTheme(theme === "light" ? "dark" : "light")}
   430→                aria-label="Toggle theme"

   432→                {theme === "light" ? <Moon className="h-4 w-4" /> : <Sun className="h-4 w-4" />}

   434→
   435→              {viewer ? (
   436→                <div className="flex items-center gap-2 rounded-full border border-border/60 pl-1 pr-3 py-1 bg-card">

   459→            </div>
   460→          </div>
   461→        </div>
   462→      </header>
   463→
   464→      <main className="flex-1 container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-8">
   465→        <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">

   480→            </TabsTrigger>
   481→            <TabsTrigger value="capability" disabled={!scanResult} className="py-2.5">
   482→              <Network className="h-4 w-4 mr-2" /> Capabilities

   489→          {/* SETUP TAB */}

   556→              {/* Step 2: Auth */}
